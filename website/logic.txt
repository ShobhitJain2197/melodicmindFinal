song_display.html

<!DOCTYPE html>
<html>
<head>
    <title>{{ song_name }}.txt</title>
    <link rel="stylesheet" href="/static/style.css">
    <script>
        let currentLine = 0;

        function startScroll() {
            const interval = setInterval(async () => {
                try {
                    const res = await fetch("/listen");
                    const data = await res.json();
                    if (data.scroll) {
                        const lines = document.querySelectorAll('#lyrics pre');
                        if (currentLine < lines.length) {
                            lines[currentLine].scrollIntoView({ behavior: 'auto', block: 'start' });
                            currentLine++;
                            document.getElementById("status").innerText = "✅ Scrolled line " + currentLine;
                        }
                    }
                } catch (e) {
                    console.error("Error:", e);
                    clearInterval(interval);
                }
            }, 1000);
        }
    </script>
</head>
<body>
    <h1>{{ song_name }}</h1>
    <p>Play chords and scroll lyrics automatically!</p>
    <button onclick="startScroll()">▶️ Start Listening</button>
    <p id="status"></p>

    <div id="lyrics" style="padding: 30px;">
        {% for line in lines %}
            <pre>{{ line }}</pre>
        {% endfor %}
    </div>

    <p style="margin-top: 40px; font-size: 12px; color: #888;">Created by <strong>SHOBHIT JAIN</strong></p>
